Sub GetDbInfo(dbDir As NotesDbDirectory, csvPath As String)
	
    ' --- 出力項目定義
	Dim target_db_info List As String
	Call GetTargetDbInfo(target_db_info)
	
	Dim session As New NotesSession
	Dim stream As NotesStream
	Set stream = session.CreateStream
	Call stream.Open(csvPath, "UTF-8")
	Call stream.Truncate
	
    ' --- CSVヘッダ作成
	Dim header As String
	
	Forall headerName In target_db_info
		header = header & headerName & ","
	End Forall
	
	Call stream.WriteText(header & Chr(13) & Chr(10))
	
    ' --- CSVボディ作成
	Dim db As NotesDatabase
	Set db = dbDir.GetFirstDatabase(DATABASE)
	
	While Not db Is Nothing
		
		If db.IsOpen Or db.Open("", "") Then
			
			Dim headerTag As String
			Dim strLine As String
			strLine = ""
			
			
			Forall v In target_db_info
				headerTag = Listtag(v) 
				strLine = strLine & GetDbValue(db, headerTag) & ","
			End Forall
			
			Call stream.WriteText(strLine & Chr(13) & Chr(10))
		End If
		
		Set db = dbDir.GetNextDatabase
	Wend
	
	Call stream.Close
End Sub
